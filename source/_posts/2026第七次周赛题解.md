## [A1](https://ac.nowcoder.com/acm/problem/277170 )

### 核心思路

贪心。先对数组按升序排序并求出初始总和：

若初始总和小于目标值，为了用最少步数拉高总和，优先从最小的数开始，将其修改为最大上限 $10^4$；
<!-- more -->
若初始总和大于目标值，为了用最少步数拉低总和，优先从最大的数开始，将其修改为最小下限 $-10^4$。

每次修改后判断总和是否跨过目标值，跨过即代表可以通过调整当前数字为某一个界内值来精确匹配，此时直接输出操作次数。

### AC代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
    
const int N = 1e4;
    
void solve(){
    
    int n; cin >> n;
    int ans; cin >> ans;
    vector<int> a(n + 1);
    int sum = 0, cnt = 0;
    for (int i = 1; i <= n; ++i) {
        int x; cin >> x;
        a[i] = x;
        sum += x;
    }
    sort(a.begin() + 1, a.end());
    if (ans == sum) {
        cout << 0 << endl;
    }
    else if(ans > sum){
        for (int i = 1; i <= n; ++i) {
            ++cnt;
            sum -= a[i];
            sum += N;
            if (sum >= ans) {
                cout << cnt << endl;
                break;
            }
        }
    }
    else {
        for (int i = n; i >= 1; --i) {
            ++cnt;
            sum -= a[i];
            sum -= N;
            if (sum <= ans) {
                cout << cnt << endl;
                break;
            }
        }
    }
    
}
    
signed main(){
    
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int _ = 1;
    cin >> _;
    while (_--)
    solve();
    return 0;
    
}
```

## [A2](https://ac.nowcoder.com/acm/problem/289292)

### 核心思路

构造法。若 $x$ 的位数为 $k$，只需构造 $y = 10^k + 1$。

此时 $x \times y = x \cdot 10^k + x$，在十进制表示下相当于将 $x$ 的数字序列原样拼接了两次（例如 $x$ 为 13 时，$13 \times 101 = 1313$）。这样 $x \times y$ 的数集与 $x$ 的数集完全相等，必然满足是其非空子集的条件。

### AC代码

```cpp
#include<bits/stdc++.h>
#define int long long
using ll = long long;
using namespace std;
    
const int N = 1e5+5;
    
void solve(){
    
    int n; cin >> n;
    int ans = 1;
    while (n) {
        n /= 10;
        ans *= 10;
    }
    cout << ans + 1 << endl;
    
}
    
signed main(){
    
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int _ = 1;
    cin >> _;
    while (_--)
    solve();
    return 0;
    
}
```

## [A3](https://ac.nowcoder.com/acm/problem/308137)

### 核心思路

对一段区间内的元素乘以 $k$，等同于让整个数组的总和增加了“区间和 $\times (k-1)$”。由于给定条件中 $k \ge 1$，要使最终的总和最大，只需求出原数组的**最大连续子段和**即可。利用前缀和，在遍历时动态维护出现过的最小前缀和，以 $O(n)$ 复杂度求出最大子段和，最终结果即为“原总和 $+$ 最大子段和 $\times (k-1)$”。

### AC代码

```cpp
#include<bits/stdc++.h>
#define int long long
using ll = long long;
using namespace std;
    
const int N = 1e5+5;
    
void solve(){
    
    int n, k; cin >> n >> k;
    vector<int> a(n + 1), pre(n + 1);
    int mn = 0, mx = -1;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        pre[i] = pre[i - 1] + a[i];
        mx = max(pre[i] - mn, mx);
        mn = min(pre[i], mn);
    }

    cout << pre[n] + mx * (k - 1) << endl;
    
}
    
signed main(){
    
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int _ = 1;
    // cin >> _;
    while (_--)
    solve();
    return 0;
    
}
```

## [B1](https://ac.nowcoder.com/acm/problem/293617)

### 核心思路

Saki 的最优策略是沿着距离起点 $k$ 最长的路径逃跑。由于 Saki 可以在每次移动后放置替身，她可以不断诱导 Miku，确保自己能安全到达离起点最远的节点。因此，游戏能坚持的最大回合数，就等于以起点 $k$ 为根节点时，树的最大深度（即距起点 $k$ 的最远距离）加 $1$。通过一次 BFS 或 DFS 求出该最大距离即可。

### AC代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
    
void solve(){
    
    int n, k; 
    cin >> n >> k;
    
    vector<vector<int>> adj(n + 1);
    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    int max_dist = 0;
    queue<tuple<int, int, int>> q;
    q.push({k, -1, 0});
    
    while (!q.empty()) {
        auto [curr, parent, dist] = q.front();
        q.pop();
        
        max_dist = max(max_dist, dist);
        
        for (int neighbor : adj[curr]) {
            if (neighbor != parent) {
                q.push({neighbor, curr, dist + 1});
            }
        }
    }
    
    cout << max_dist + 1 << "\n";
    
}
    
signed main(){
    
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int _ = 1;
    // cin >> _;
    while (_--)
        solve();
    return 0;
    
}
```

## [B2](https://www.luogu.com.cn/problem/P2820)

### 核心思路

本题本质上是求**最小生成树**（MST）。为了在保持图连通的前提下消除回路，且使得被删去的边权总和最大，就需要让保留下来的边权总和最小。利用 Kruskal 算法配合并查集求出最小生成树，答案即为“原图所有边权总和”减去“最小生成树的边权总和”。

### AC代码

```cpp
#include<bits/stdc++.h>
#define int long long
using ll = long long;
using namespace std;
    
const int N = 1e5+5;

struct Edge {
    int u, v, w;
    bool operator< (const Edge& o) const{
        return w < o.w;
    }
};

struct DSU {
    vector<int> fa;

    DSU(int n) : fa(n + 1) {
        iota(fa.begin(), fa.end(), 0);
    }

    int find(int x) {
        return fa[x] == x ? x : fa[x] = find(fa[x]);
    }

    bool merge(int u, int v) {
        int ru = find(u), rv = find(v);
        if (ru == rv) return false;
        fa[ru] = rv;
        return true;
    };
};

void solve(){
    
    int n, k; cin >> n >> k;
    vector<Edge> a(k + 1);

    int sum = 0, e = 0;
    for (int i = 1; i <= k; ++i) {
        cin >> a[i].u >> a[i].v >> a[i].w;
        sum += a[i].w;
    }

    sort(a.begin() + 1, a.end());
    
    DSU dsu(n);
    int tsum = 0, te = 0;
    for (int i = 1; i <= k; ++i) {
        if (dsu.merge(a[i].u, a[i].v)) {
            tsum += a[i].w;
        }
    }

    cout << sum - tsum << endl;

}
    
signed main(){
    
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int _ = 1;
    //cin >> _;
    while (_--)
    solve();
    return 0;
    
}
```

## [C](https://ac.nowcoder.com/acm/problem/297762)

### 核心思路

题目要求count("01")+count("10")==3，也就是说明存在三个位置s[i]!=s[i-1]

很明显，把整个字符串变成0101或者1010就行了(注意：这里不是说要01连续，而是整段的0和整段的1的组成字符串)

### AC代码

```python
import sys
input=sys.stdin.readline
# 写的是py，凑合着看
n=int(input())
s=input().strip()
pre0,pre1=[0]*n,[0]*n
pre0[0]=(1 if s[0]=='0' else 0)
pre1[0]=(0 if s[0]=='0' else 1)
for i in range(1,n):
    pre1[i]=(1+pre1[i-1] if s[i]=='1' else pre1[i-1])
    pre0[i]=(pre0[i-1] if s[i]=='1' else pre0[i-1]+1)
ans=float('inf')
for i in range(0,n):
    for j in range(i+1,n):
        for z in range(j+1,n-1):
            ans=min(pre0[i]+pre1[j]-pre1[i]+pre0[z]-pre0[j]+pre1[n-1]-pre1[z],ans)
            ans=min(pre1[i]+pre0[j]-pre0[i]+pre1[z]-pre1[j]+pre0[n-1]-pre0[z],ans)
print(ans)
```

## [D](https://ac.nowcoder.com/acm/problem/219918)

### 核心思路

极端情况下（如当大量的数在某一位上均为 $1$ 时）原图会退化成完全图，边数达到 $O(n^2)$ 级别。此时使用常规 Dijkstra 算法扩展边，时间复杂度会爆炸导致超时。

**建虚点优化（AC代码）**：

条件 $a_i \& a_j \neq 0$ 的本质是：数 $a_i$ 和 $a_j$ 在二进制表示下，至少存在一个相同的位都是 $1$。我们可以将这种“两两直接连边”转化为“通过公共枢纽相连”。

1. **建立虚点**：额外增加 31 个虚拟节点，分别代表二进制的第 $0 \sim 30$ 位。
2. **把边拆成两半**：题目要求 $i \to j$ 的边权是 $a_i + a_j$，我们将这条边拆成两半，把权值分摊到进入和离开虚点的过程上：
   - **入边**：如果 $a_i$ 的第 $p$ 位是 $1$，则从真实节点 $i$ 走向虚拟节点 $p$，花费代价 $a_i$。
   - **出边**：如果 $a_j$ 的第 $p$ 位也是 $1$，则从虚拟节点 $p$ 可以走向真实节点 $j$，花费代价 $a_j$。
3. **等价代换**：原本的路径 $i \to j$ 就变成了 $i \to \text{虚拟节点 } p \to j$，此时走过的总距离恰好就是 $a_i + a_j$。

通过这种建图方式，我们将原本 $O(n^2)$ 的边数降维到了每个点最多只与 31 个虚点相连。总边数控制在 $O(31 \times n)$ 级别。

### AC代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
    
const int MAXN = (1LL << 60);

struct Edge {
    int d, u;
    bool operator< (const Edge& o) const {
        return d > o.d;
    }
};
    
void solve(){
    
    int n; cin >> n;
    vector<int> a(n + 1), dis(n + 35, MAXN), vis(n + 35);
    vector<int> g[35];
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        for (int p = 0; p <= 30; ++p) {
            if ((a[i] >> p) & 1) {
                g[p].push_back(i);
            }
        }
    }

    priority_queue<Edge> pq;
    pq.push({dis[1] = 0, 1});

    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();

        if (vis[u]) continue;
        vis[u] = 1;

        if (u <= n) {
            for (int p = 0; p <= 30; ++p) {
                if ((a[u] >> p) & 1) {
                    int bitnode = n + p + 1;
                    if (dis[bitnode] > dis[u] + a[u]) {
                        pq.push({dis[bitnode] = dis[u] + a[u], bitnode});
                    }
                }
            }
        }

        else {
            int p = u - n - 1;
            for (int v: g[p]) {
                if (dis[v] > dis[u] + a[v]) {
                    pq.push({dis[v] = dis[u] + a[v], v});
                }
            }
        }

    }

    for (int i = 1; i <= n; ++i) {
        if (dis[i] == MAXN) cout << -1 << " ";
        else cout << dis[i] << " ";
    }
    cout << "\n";

}
    
signed main(){
    
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int _ = 1;
    // cin >> _;
    while (_--)
    solve();
    return 0;
    
}
```
