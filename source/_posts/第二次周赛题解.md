#include<bits/stdc++.h>
#define int long long
using ll = long long;
using namespace std;
    
const int N = 1e5+5;

    int cnt = 0;
    
    while (a > 0 && b > 0) {
        if (a == 1 && b == 1) {

        }
        else if (a < b) {
            a++;
            b-=2;
        }
        else {
            a-=2;
            b++;

    }
    
    cout << cnt << endl;
}

### B1

<hr style="border:none; border-top:1px dotted #555;" />

 **tags** 

* [abc407_c](https://atcoder.jp/contests/abc407/tasks/abc407_c )
* 难度1100
* 暴力、模拟、贪心



正向思考时，每次按按钮 B 都会影响当前所有的数字，这让计算变得复杂。但如果我们**从最后一个字符开始倒着考虑**，逻辑就会变得非常清晰。 

1. 每个字符的生成必须按一次 **按钮 A**。所以，最终答案中必然包含 $n$（字符串长度）次按钮 A 的操作。 

2. 当你为了让 最后一个被加入的字符 达到目标值而按下 $k$ 次按钮 B 时，**在它之前已经存在的所有字符**也都被动地增加了 $k$。 所以，我们从最后一位倒推，还原最后一个字符被加入之前的字符串。

暴力写法（$O(N^2)$，过不了）：

```cpp
void solve(){ 
    string s; cin >> s;
    int n = s.length();
    vector<int> a(n + 1);
    for (int i = 0; i < n; ++i) {
        a[i + 1] = s[i] - '0';
    }
    
    int ans = 0;
    for (int i = n; i >= 1; --i) {
        
        ans += a[i]; // 当前位需要按 a[i] 次 B
        
        for (int j = 1; j <= i - 1; ++j) {
            // 前面的位被动增加了 a[i]，现在还原回去
            a[j] = (a[j] - a[i] + 10) % 10; 
        }
    }

    cout << ans + n << endl; // 总次数 = 按钮 B 次数 (ans) + 按钮 A 次数 (n)
}
```



我们发现，前面某一位 $a[i-1]$ 最终被动增加了多少次，其实就是它后面所有位操作次数的总和。 

优化写法 （$O(N)$） ：

```cpp
void solve(){
    string s; cin >> s;
    int n = s.length();
    vector<int> a(n + 1);
    for (int i = 0; i < n; ++i) {
        a[i + 1] = s[i] - '0';
    }
    
    int ans = 0;
    for (int i = n; i >= 1; --i) {
        ans += a[i];
        
        // 更新前一位：它受到的总影响就是目前为止所有的 ans
        a[i - 1] = a[i - 1] - ans % 10;
        a[i - 1] = (a[i - 1] + 10) % 10;
    }
    
    cout << ans + n << endl;
}
```

### B2

<hr style="border:none; border-top:1px dotted #555;" />

 **tags** 

* [洛谷P5002](https://www.luogu.com.cn/problem/P5002 )
* 难度1200
* dfs、枚举

这题虽然题目背景是 LCA(最近公共祖先)，但只需要会求子树大小就能过 (=・ω・=)

题目让我们数有多少个点对(u, v)，使它们的最近祖先都是 $P$。 

1. 设 $sz[P]$ 表示以 $P$ 为根的子树节点总数。 

2. 若 $P$ 没有儿子， 显然，就在 $P$ 子树内选两个点（有序且可重复），共有 $sz[P] \times sz[P]$ 种可能。 

3. 若 $P$ 有儿子， 我们需要排除那些最近公共祖先不是 $P$ 的情况：

   * 如果选出的两个点 $x$ 和 $y$ 同时位于**$P$ 的儿子 $$Q$$** 的子树内，那么它们的 最近公共祖先 一定是 $Q$ 或 $Q$ 的后代，而不是 $P$。 

4. 所以我们把 $$P$$ 的儿子的子树排除，得到最终公式：
   $$
   ans(P) = sz[P]^2 - \sum_{Q \in son(P)} sz[Q]^2
   $$

代码如下：

```cpp
vector<int> adj[N];
int sz[N], fa[N];

void dfs(int u, int f, int d) {
    sz[u] = 1;
    fa[u] = f;
    for (int v: adj[u]) {
        if (v == f) continue;
        dfs1(v, u, d + 1);
        sz[u] += sz[v];
    }
}

void solve(){
    
    int n, m, r; cin >> n >> r >> m;
    for (int i = 1; i < n; ++i) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    dfs(r, 0, 1);

    while (m--) {
        int u; cin >> u;
        int ans = sz[u] * sz[u];
        for (int v: adj[u]) {
            if (v == fa[u]) continue;
            ans -= sz[v] * sz[v];
        }
        cout << ans << endl;
    }
    
}
```

异或是指两个bool变量如果相同返回false，不同返回true
按位异或()的性质：
1.按位异或是对于两个数的每一个二进制位，如果全为0或1，则该位异或值为0，否则为1
2.容易得到一个数按位异或0等于这个数本身
3.可以看出，对于a按位异或x=b，如果a和b已知，x的值是唯一的

本题可以先打表：
用f(x)表示答案
可以得到f(1)=1,f(2)=1,f(3)=4,f(4)=4,f(5)=8,...,f(8)=8,f(9)=16,...
可以看出对于后面(n>=3)的2^k(2的k次方，后面同理)到2^(k+1)-1的一段有
2^k异或x=2^k，x只能是0，故无法得到2^k，答案就是2^k
而2^k和其他数异或可以得到0至2^k-1的所有结果
2^k+1因为2^(k+1)异或1=2k，这样可以得到2^(k+1)-1以内的所有结果，答案为2^(k+1)
其余的由于更高一位的都是0，所以无法突破2^(k+1)的范围，所以答案为2^(k+1)

对于n==1或2，需要特判
时间复杂度为O(Tlog(n))

### D1

<hr style="border:none; border-top:1px dotted #555;" />

 **tags** 

*  [CF2086D](https://codeforces.com/problemset/problem/2086/D) 
*  难度1700
*  动态规划、组合数学

可以看出同一字母要么都在单号，要么都在双号
设单号有p个，双号有q个，p为n/2向上取整，q为n/2向下取整，n为c[i]的总和

这样分开考虑，对于单号某一次排满，数量为A(p,p)/PI((单号各自数量)!),双号的某一次排满，数量为A(q,q)/PI((双号各自数量)!)
前面两种情况相乘即为某一次排满的数量
此时你会惊奇的发现，正好为A(p,p)*A(q,q)/PI(c[i](i=1~26)!)，为固定值，而该值乘排满的次数就是答案。

对于排满的次数，可以参考01背包问题，使用dp求解
设dp[x]为考虑前i个数，能组成和恰好为x的次数，并对P=998244353取模
转移方程如下(已做压维和滚动数组优化处理)
for(int i=1;i<=26;i++){
    for(int j=p;j>=c[i];j--){
        dp[j]=(dp[j]+dp[j-c[i]])%P;
    }
}
这样求出的dp[p]就是想要的排满的次数
最后注意对于可能出现爆long long的地方取模，这样就可以得出答案了。
时间复杂度为O(26*n)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const ll P = 998244353;

ll qmi(ll a, ll x){
    ll res = 1;
    while(x > 0) {
        if(x & 1) res = res * a % P;
        a = a * a % P;
        x >>= 1;
    }
    return res;
}

ll inv(ll a) {
    return qmi(a, P-2);
}

ll fact[500008];

void solve(){
    vector<int> c(27,0);
    int n = 0;
    ll PI26 = 1ll;
    for(int i = 1; i <= 26; i++){
        cin >> c[i];
        n += c[i];
        PI26 = PI26 * fact[c[i]] % P;
    }
    int odd = (n + 1) / 2;
    int even = n / 2;
    vector<ll> dp(n+1, 0);
    dp[0] = 1;
    for(int i = 1; i <= 26; i++){
        if(c[i] == 0){
            continue;
        }
        for(int j = n; j >= c[i]; j--){
            dp[j] = (dp[j - c[i]] + dp[j]) % P;
        }
    }

    ll ans = fact[odd] * fact[even] % P * inv(PI26) % P * dp[odd] % P;
    cout << ans << endl;
}

int main(){
    fact[0]=1;
    for(ll i=1; i <= 500000; i++){
        fact[i] = fact[i - 1] * i % P;
    }
    int t;
    cin >> t;
    while(t--) {
        solve();
    }
}
```
