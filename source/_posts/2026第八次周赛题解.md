# [A]( https://atcoder.jp/contests/abc114/tasks/abc114_b )

从每一位开始枚举，枚举3位后与答案汇总，得出最接近的值。
<!--more-->

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    string s; cin >> s;
    int ans = 1e9+7;
    for(int i = 0;i<s.length()-2;i++){
        int t = 0;
        for(int j = 0;j<3;j++){
            t = t * 10 + s[j+i]-'0';
        }
        ans = min(ans , abs(t-753));
    }
    cout << ans << endl;
    
    return 0;
}
```

时间复杂度：$O(n)$

# [B]( https://ac.nowcoder.com/acm/contest/11216/E )

按照题意操作即可，这里给出位运算版本

- 位运算

```cpp
#include <iostream>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n;
    cin >> n;
    long long total = 0;
    
    for (int i = 0; i < n; ++i) {
        int a;
        cin >> a;
        long long reversed = 0;
        int x = a;
        
        while (x > 0) {
            reversed = (reversed << 1) | (x & 1);
            x >>= 1;
        }
        
        total += reversed;
    }
    
    cout << total << '\n';
    return 0;
}
```

时间复杂度：$O(nlogA)$

# [C]( https://ac.nowcoder.com/acm/contest/99458/C )

这三种操作中，直觉上开平方的数字下降幅度最大。打表或者数学推导可以得到：

- 当$n \geq 3$ 时，开平方的下降幅度大于除2和减1
- 当$ 0 \leq n \leq 3$ 时，减1和除2下降幅度相等
- 当$n \leq 0$时，显然无法开平方，除2会导致答案增加，故选择减1

可以得到如下推导式：
$$
\begin{align}
&= \sqrt n, n \geq 3 \\
n \\
&= n-1 , n < 3
\end{align}
$$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    
    ll n,m;
    cin>>n>>m;
    while(m){
        m--;
        if(n>=3){
            n=ceil(sqrt(n));
        }else{
            n=n-1;
        }
        if(n==0){
            break;
        }
    }
    cout<<n-m<<"\n";
}
```

时间复杂度：$O(\sqrt n)$

# [D]( https://www.luogu.com.cn/problem/P1661 )

二维平面上给定 n 个点， 两点连通当且仅当它们的 曼哈顿距离 ≤ 2t

 求最小的 t， 使得所有点在该连通关系下构成一个连通块。 

- 曼哈顿距离：对于二维平面上的两点$A(x1,y1),B(x2,y2)$，其曼哈顿距离定义为横坐标之差的绝对值+纵坐标之差的绝对值，即$dist(A,B) = |x1-x2| + |y1-y2|$

## 二分+并查集

假设$x$是最小的满足条件的$t$，那么大于$x$的值都将满足条件，小于$x$的值都将不满足条件，可以看出，答案具有单调性，故采用二分。

将两点之间的曼哈顿距离看作这两点连成的边的边权，如果在曼哈顿距离都满足$\leq 2x$的情况下能合并为一个连通块，则满足条件，否则不满足。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

vector<int> f(2005);

int find(int x){
    return f[x] = f[x]==x ? f[x] : find(f[x]);
}

void solve()
{
    int n; cin >> n;
    vector<pair<int,int> > vp(1);
    for(int i = 1;i<=n;i++){
        int u,v; cin >> u >> v;
        vp.push_back({u,v});
    }

    auto dist = [&](pair<int,int>& a,pair<int,int>& b)->int{
        return abs(a.first-b.first)+abs(a.second-b.second);
    };

    auto check = [&](int x)->int{
        for(int i = 1;i<=n;i++){
            f[i] = i;
        }

        int cnt = 0;
        for(int i = 1;i<=n;i++){
            for(int j = i+1;j<=n;j++){
                int fx = find(i),fy = find(j);
                if(fx!=fy && dist(vp[i],vp[j])<=x*2){
                    f[fx] = fy;
                    cnt++;
                }
            }
        }
        return cnt==n-1;
    };

    ll l = 0,r = 2e9,mid , ans = 0;
    while(l<=r){
        mid = l+r>>1;
        if(check(mid)){
            r = mid - 1;
            ans = mid;
        }
        else{
            l = mid + 1;
        }
    }
    cout << ans << endl;
}

int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    ll _ = 1;
    // cin >> _;
    while (_--)
    {
        solve();
    }
}
```

时间复杂度：$O(n^2logA)$

## kruskal最小生成树

问题等价于求该图的

最小瓶颈生成树（Minimum Bottleneck Spanning Tree） 的瓶颈值（即 MST 中的最大边权

记录所有的边，按照边权从小到大排序，然后使用kruskal算法得到答案

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

vector<int> f(2005);

int find(int x){ return f[x] = f[x]==x ? f[x] : find(f[x]);}

void solve()
{
    int n; cin >> n;
    vector<pair<ll,ll> > vp(n+1);
    for(int i = 1;i<=n;i++){
        cin >> vp[i].first >> vp[i].second;
        f[i] = i;
    }
    
    auto dist = [&](pair<ll,ll>& a,pair<ll,ll>& b)->ll{
        return abs(a.first-b.first)+abs(a.second-b.second);
    };
    
    vector<array<ll,3> > e;
    for(int i = 1;i<=n;i++){
        for(int j = i+1;j<=n;j++){
            e.push_back({i,j,dist(vp[i],vp[j])});
        }
    }
    sort(e.begin(),e.end(),[](auto a,auto b){
        return a[2]<b[2];
    });
    
    int cnt = 0;
    for(auto [i,j,w] : e){
        int fx = find(i),fy = find(j);
        if(fx!=fy){
            f[fx] = fy;
            cnt++;
            if(cnt==n-1) return [&]{cout << (w+1)/2 << endl;}();
        }
    }
    
}

int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    ll _ = 1;
//     cin >> _;
    while (_--)
    {
        solve();
    }
}
```

时间复杂度：$O(n^2logn)$

# [E]( https://www.luogu.com.cn/problem/P1629 )

给定一个有向图，求从1节点出发到其他节点的最短路径之和+从其他节点出发到1节点的最短路径之和

这里只给最优解，原题数据可以跑到$O(nmlogn)$

- 求从某个节点出发到其他节点的最短路径，可以通过单源最短路算法实现，如dijkstra，SPFA等
- 从其他节点到某个节点的最短路径，相当于从某个节点反向通过边到达其他节点，所以第二部分的答案可以通过建反图处理

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll N = 1e5+10;
const ll inf = 1e13+7;

struct node{
    ll to,next,w;
}e[N],re[N];
int head[N],rhead[N];
int cnt = 0,rcnt = 0;

void add(int u,int v,int w){
    cnt++;
    e[cnt].to = v;
    e[cnt].next = head[u];
    head[u] = cnt;
    e[cnt].w = w;
}

void radd(int u,int v,int w){
    rcnt++;
    re[rcnt].to = v;
    re[rcnt].next = rhead[u];
    rhead[u] = rcnt;
    re[rcnt].w = w;
}

void solve()
{
    int n,m; cin >> n >> m;
    for(int i = 1;i<=m;i++){
        int u,v,w; cin >> u >> v >> w;
        add(u,v,w);
        radd(v,u,w);
    }

    vector<ll> dist(n+1,inf),rdist(n+1,inf);
    dist[1] = rdist[1] = 0;
    priority_queue<pair<ll,ll> , vector<pair<ll,ll>> ,greater<>> q;
    q.push({0,1});
    vector<int> vis(n+1,0);
    while(q.size()){
        auto [val,t] = q.top();
        q.pop();

        if(vis[t]) continue;
        vis[t] = 1;
        for(int i = head[t];i != 0;i = e[i].next){
            int v = e[i].to;
            ll w = e[i].w;
            if(dist[v]>=val+w){
                dist[v] = val + w;
                q.push({dist[v],v});
            }
        }
    }

    q.push({0,1});
    vis.assign(n+1,0);
    while(q.size()){
        auto [val,t] = q.top();
        q.pop();

        if(vis[t]) continue;
        vis[t] = 1;
        for(int i = rhead[t];i != 0;i = re[i].next){
            int v = re[i].to;
            ll w = re[i].w;
            if(rdist[v]>=val+w){
                rdist[v] = val + w;
                q.push({rdist[v],v});
            }
        }
    }
    ll sum = 0;
    for(int i = 1;i<=n;i++){
        sum += dist[i]+rdist[i];
    }
    cout << sum << endl;
}

int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    ll _ = 1;
    // cin >> _;
    while (_--)
    {
        solve();
    }
}
```

时间复杂度：$O(mlogn)$

# [F]( https://codeforces.com/problemset/problem/1633/D )

给定操作，在k次操作之后，求最多能获得多少收益。

这题考察对数据范围的敏感度。

**注意到** $n \leq 1000 , b_i \leq 1000$，  可以$O(n^2)$预处理至少需要多少次才能让$a_i$从1变成$b_i$。

将从1变成大于1的数所需的操作次数打表之后得出，对单个值最大的操作次数**不超过12**

所以可以把k的上限从$10^6$下调到最多$12n$

最后就是经典的01背包了。~~（疑似这题最简单的部分）~~

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll p[2005]; 

void solve()
{
    ll n, k;
    cin >> n >> k;
    vector<int> b(n + 1, 0), c(n + 1, 0);
    for (int i = 1; i <= n; i++)
    {
        cin >> b[i];
        b[i] = p[b[i]];
    }
    for (int i = 1; i <= n; i++)
    {
        cin >> c[i];
    }

    k = min(k, 12 * n);
    vector<vector<ll>> dp(n + 1, vector<ll>(k + 1, 0));
    for (int i = 1; i <= n; i++)
    {
        int v = b[i], w = c[i];
        for (int j = 0; j <= k;j++){
            dp[i][j] = dp[i - 1][j];
            if(j>=v)
                dp[i][j] = max(dp[i][j], dp[i - 1][j - v] + w);
        }
    }
    cout << dp[n][k] << endl;
}

int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    for (int i = 2; i <= 2000; i++)
    { 
        p[i] = 1e9 + 7;
    }
    p[1] = 0;
    for (int i = 1; i <= 1000; i++)
    {
        for (int j = 1; j <= i; j++) // 当j取大于i的值时，无意义-> i/j==0
        {
            p[i + (i / j)] = min(p[i + (i / j)], p[i] + 1);
        }
    }

    int _ = 1;
    cin >> _;
    while (_--)
    {
        solve();
    }
}
```

时间复杂度：$O(n^2)$

# [G]( https://codeforces.com/contest/1985/problem/H2 )

给定二维网格，网格上分布由'#'组成的连通块，求将任意一行**和**一列变为'#'之后所形成的可能的最大连通块的大小。

我们假定，最大可能的连通块一定是我们变化后的连通块。如果变化后还有其他更大的连通块，那么我们可以选择那个更大的连通块作为我们答案的一部分。

假设我们只将一行变为'#'。

- 不考虑联通其他块，其对答案的贡献为当前行'.'的数量。
- 只考虑其他连通块，假设这是第$j$行，那么对答案有贡献的连通块即为第$j-1,j,j+1$行有'#'的连通块的大小。

对于情况1，数格点即可预处理；情况2可以反过来，预处理出单个连通块对可能做出贡献的区间的贡献，最小边界-1到最大边界+1为贡献区间。边界和连通块大小可以用洪水填充算法解决。

只变一列也同理。

注意不要把行和列同时触碰的连通块重算，所以再预处理重复的范围，数点也一样。

这些信息可以用前缀和差分维护。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

void solve()
{
    int n, m;
    cin >> n >> m;
    vector<vector<int>> g(n + 1, vector<int>(m + 1, 0));
    for (int i = 1; i <= n; i++)
    {
        string s;
        cin >> s;
        s = ' ' + s;
        for (int j = 1; j <= m; j++)
        {
            g[i][j] = s[j] == '#';
        }
    }

    vector<int> sumrow(n + 3, 0), sumcol(m + 3, 0);
    vector<int> addrow(n + 1, 0), addcol(m + 1, 0);
    vector<vector<int>> vis(n + 1, vector<int>(m + 1, 0));
    vector<vector<int>> sum(n + 3, vector<int>(m + 3, 0));
    int maxC, minC, maxR, minR, sz;
    const int d[5] = {0, -1, 0, 1, 0};

    auto check = [&](int x, int y) -> int
    {
        return x > 0 && x <= n && y > 0 && y <= m && !vis[x][y] && g[x][y];
    };

    auto dfs = [&](auto self, int x, int y) -> void
    {
        vis[x][y] = 1;
        sz++;

        minC = min(minC, y);
        maxC = max(maxC, y);
        minR = min(minR, x);
        maxR = max(maxR, x);

        for (int i = 0; i < 4; i++)
        {
            int xx = d[i] + x, yy = d[i + 1] + y;
            if (check(xx, yy))
                self(self, xx, yy);
        }
    };

    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= m; j++)
        {
            if (!vis[i][j] && g[i][j])
            {
                minC = minR = 1e9 + 7;
                maxC = maxR = 0;
                sz = 0;
                dfs(dfs, i, j);

                sumrow[maxR + 2] -= sz;
                sumrow[minR - 1] += sz;
                sumcol[minC - 1] += sz;
                sumcol[maxC + 2] -= sz;
                sum[maxR + 2][maxC + 2] += sz;
                sum[maxR + 2][minC - 1] -= sz;
                sum[minR - 1][maxC + 2] -= sz;
                sum[minR - 1][minC - 1] += sz;
            }
            if (!g[i][j])
            {
                addrow[i]++;
                addcol[j]++;
            }
        }
    }

    for (int i = 1; i <= n; i++)
    {
        sumrow[i] += sumrow[i - 1];
    }
    for (int j = 1; j <= m; j++){
        sumcol[j] += sumcol[j - 1];
    }
    
    for (int i = 1; i <= n + 2;i++){
        sum[i][0] += sum[i - 1][0];
    }
    for (int j = 1; j <= m + 2;j++){
        sum[0][j] += sum[0][j - 1];
    }
    
    for (int i = 1; i <= n + 2; i++)
    {
        for (int j = 1; j <= m + 2; j++)
        {
            sum[i][j] = sum[i][j] + sum[i - 1][j] + sum[i][j - 1] - sum[i - 1][j - 1];
        }
    }

    int ans = 0;

    for (int i = 1; i <= n;i++){
        for (int j = 1; j <= m;j++){
            ans = max(ans, sumrow[i] + sumcol[j] - sum[i][j] + addrow[i] + addcol[j] - !g[i][j]);
        }
    }

    cout << ans << endl;
}

int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    ll _ = 1;
    cin >> _;
    while (_--)
    {
        solve();
    }
}
```

时间复杂度：$O(nm)$
