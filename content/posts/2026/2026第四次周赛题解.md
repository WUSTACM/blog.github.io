## [A2](https://atcoder.jp/contests/abc229/tasks/abc229_c)

注意读题，不要写成01背包，二进制优化什么的了，它就是一道简单的 **贪心**

大家可以尝试一下`array`，很好用的

注意开`long long`

~~~C++
using arr2 = array<int, 2>;
void solve() {
    int n, w;
    cin >> n >> w;
    vector<array<ll, 2>>a(n + 5);
    for (int i = 1; i <= n; i++) {
        cin >> a[i][0] >> a[i][1];
    }
    ll ans = 0;
    sort(a.begin() + 1, a.begin() + 1 + n);
    for (int i = n; i >= 1; i--) {
        ans += a[i][0] * min(1ll * w, a[i][1]);
        w -= min(1ll * w, a[i][1]);
        if (!w) break;
    }
    cout << ans << '\n';
}
~~~



## [A3](https://atcoder.jp/contests/abc230/tasks/abc230_b)

改了数据，加了多测

注意它是 $10^5$ 个`oxx` 拼接起来的，所以我们可以先把这个大字符串搞出来，再用`string` 的`find` 就可以了

如果是 $10 ^ {1000}$   个`oxx` 拼接起来的呢，又要怎么写？

我们考虑合法的 `S` 长什么样子，它一定是 `oxx` 的一段后缀开头然后接着`oxx`，所以我们枚举出合法的`S`就可以了

~~~C++
void solve() {
    string s, t = "oxx"; 
    cin >> s;
    int n = s.size();
    vector<string>ans;
    for (int pos = 0; pos < 3; pos++) {
        string c = ""; 
        for (int i = 0; i < n; i++) {
            c += t[(i + pos) % 3];
        }
        ans.push_back(c);
    }
    for (string c : ans) {
        if (s == c) {
            cout << "Yes" << '\n';
            return ;
        }
    }
    cout << "No" << '\n';
}

~~~



## [B1](https://ac.nowcoder.com/acm/contest/85687/D)

$ax^2+bx+c$ 分解为 $(a_1x+b_1)(a_2x+b_2)$ 

则有$a_1 * a_2 = a$   $b_1 * b_2 = c$    $a_1 * b_2 + a_2 * b_1 = b$

注意到 $abs(a), abs(c) \leq 10 ^ 6$  所以我们可以枚举出它们的因子，$ \leq 10 ^ 6$ 的数的因子最多不超过 300,所以时间复杂度不会爆

实际是个大模拟，细节比较多

~~~C++ 
void solve() {
    int a, b, c;
    cin >> a >> b >> c;
    if (c == 0) {
        cout << 1 << ' ' << 0 << ' ' << a << ' ' << b << '\n';
        return ;
    }
    bool f1 = (a > 0);
    bool f2 = (c > 0);
    for (int i = 1; i * i <= abs(a); i++) {
        if (a % i) continue;
        for (int j = 1; j * j <= abs(c); j++) {
            if (c % j) continue;
            int a_1 = abs(i), a_2 = abs(a / i);
            int b_1 = abs(j), b_2 = abs(c / j);
            for (int m = 0; m < 4; m++) {
                if (m & 1) {
                    swap(a_1, a_2);
                }
                if (m >> 1 & 1) {
                    swap(b_1, b_2);
                }
                for (int k = 0; k < 2; k++) {
                    for (int l = 0; l < 2; l++) {
                        int a1 = 0, a2 = 0, b1 = 0, b2 = 0;
                        if (f1) {
                            if (k) {
                                a1 = -a_1;
                                a2 = -a_2;
                            }
                            else {
                                a1 = a_1;
                                a2 = a_2;
                            }
                        }
                        else {
                            if (!k) {
                                a1 = -a_1;
                                a2 = a_2;
                            }
                            else {
                                a1 = a_1;
                                a2 = -a_2;    
                            }
                        }
                        if (f2) {
                            if (l) {
                                b1 = -b_1;
                                b2 = -b_2;
                            }
                            else {
                                b1 = b_1;
                                b2 = b_2;
                            }
                        }   
                        else {
                            if (!l) {
                                b1 = -b_1;
                                b2 = b_2;
                            }
                            else {
                                b1 = b_1;
                                b2 = -b_2;    
                            }
                        }
                        if (a1 * b2 + a2 * b1 == b) {
                            cout << a1 << ' ' << b1 << ' ' << a2 << ' ' << b2 << '\n';
                            return ;
                       }
                    }
                }
                if (m & 1) {
                    swap(a_1, a_2);
                }
                if (m >> 1 & 1) {
                    swap(b_1, b_2);
                }
            }
        }
    }
    cout << "NO" << '\n';
}
~~~



事实上，这样写非常不优雅

大家还记得  **[韦达定理](https://zh.wikipedia.org/zh-cn/%E9%9F%A6%E8%BE%BE%E5%AE%9A%E7%90%86)** 吗

我们令 $A = a_1 b_2$，$B = a_2 b_1$

$A + B = a_1 b_2 + a_2 b_1 = b$

$A \times B = (a_1 b_2)(a_2 b_1) = (a_1 a_2)(b_1 b_2) = a \times c$

$A$ 和 $B$ 本质上就是一元二次方程 $t^2 - bt + ac = 0$ 的两个根

用求根公式求出 $A$ 和 $B$ 即可

我们有 $a = a_1 a_2$ 且 $A = a_1 b_2$

那么 $a_1$ 必须同时整除 $a$ 和 $A$，我们直接取最大公约数，令 $a_1 = \gcd(|a|, |A|)$

~~~C++
using ll = long long;
void solve () {
    int a, b, c;
    ll d = 0;
    cin >> a >> b >> c;
    d = b * b - 4 * a * c;
    if (d < 0) {
        cout << "NO" << '\n';
        return ;
    }
    int t = sqrtl(d);
    if (t * t != d) {
        cout << "NO" << '\n';
        return ;
    }
    if ((b + t) % 2) {
        cout << "NO" << '\n';
        return ;
    }
    ll A = (b + t) / 2;
    ll B = (b - t) / 2;

    ll a1 = gcd(abs(a), abs(A));
    ll a2 = a / a1;
    ll b1 = B / a2;
    ll b2 = A / a1;
    cout << a1 << ' ' << b1 << ' ' << a2 << ' ' << b2 << '\n';

}
~~~



> 两种方法各有利弊，你说大模拟不好写，但你在韦达定理推式子的时候，我模拟已经无脑敲完了
>
> 在赛场上能写出来AC的就是好方法



## [B2](https://codeforces.com/problemset/problem/1536/C)

题目来源：[CF1536C](https://codeforces.com/problemset/problem/1536/C)



考虑任意一个前缀的答案

该答案中划分出的任意一个子串都满足：

- 在该段中，两个字符的比例与当前这个前缀的比例相同
- 该段无法再被划分出更多的满足条件的子串

划分出的一段满足这两个条件，与其成为该前缀的答案是充要的。



若当前位置 $j$ 的前缀的比例为 $a : b$ ，上一个比例为 $a : b$ 的前缀位置是 $i$
那么不难发现从 $i + 1$ 到 $j$ 的串满足上述两个条件
那么将位置 $i$ 的答案加一就是当前位置 $j$ 的答案



考虑做法:

> 对于每个前缀，用一个对 (a,b) 标记它，其中 a = 此前缀中 'D' 的频率， b = 此前缀中 'K' 的频率。将 a 和 b 除以 gcd(a,b) 。如果我们从左到右遍历所有前缀，我们可以注意到前缀的答案等于我们到目前为止看到的这个对的出现次数
>
> 对于实现，你可以使用 C++ 中的 map 来实现 O(nlogn) 或 O(n) 的时间复杂度



代码:

```cpp
void solve () {
    int n; cin >> n;
    string s; cin >> s;
     
    map <array <int, 2>, int> mp;
    int a = 0, b = 0;
 
    for (int i = 0; i < n; i++) {
        if (s[i] == 'D') a++;
        else b++;
        int x = a / gcd(a, b);
        int y = b / gcd(a, b);
        cout << ++mp[{x, y}] << " \n"[i == n - 1];
    }
}
```

DP 数组的最小值需要满足 ![img](https://www.nowcoder.com/equation?tex=f_%7Bij%7D%20%3C%20-2%20%5Ctimes%2010%5E%7B9%2B5%7D&preview=true) 才不会被负数卡爆。

```c++
void solve(){
    int n, k; cin >> n >> k;
    vector<LL> a(n + 5);
    for(int i = 1; i <= n; ++i){
        cin >> a[i];
    }
    vector<vector<LL>> f(n + 5, vector<LL>(k + 5, -1145141919810)); // 魔法的数字，助您日入百亿
    LL R = -1145141919810;
    f[0][0] = 0;
    for(int i = 1; i <= n; ++i){
        for(int j = 1; j <= min(i, k); ++j){
            for(int p = max(0, i - 6); p <= i - 1; ++p){
                f[i][j] = max(f[i][j], f[p][j - 1] + a[i]);
            }
        }
        R = max(R, f[i][k]);
    }
    cout << R << endl;
}

```
